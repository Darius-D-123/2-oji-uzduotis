14// Should only be constructible from literal 0.
15// Based on the constructor from pointer trick, used by libstdc++ and libc++
16// (formerly also MSVC, but they've moved to consteval int constructor).
17// Used by `TypeWithLit0Comparisons` for testing comparison
18// ops that only work with literal zero, the way std::*orderings do
19struct ZeroLiteralAsPointer {
20 constexpr ZeroLiteralAsPointer( ZeroLiteralAsPointer* ) noexcept {}
21
22 template <typename T,
23 typename = std::enable_if_t<!std::is_same<T, int>::value>>
24 constexpr ZeroLiteralAsPointer( T ) = delete;
25};

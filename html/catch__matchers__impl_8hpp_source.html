32 class MatchExpr : public ITransientExpression {

35 public:
36 constexpr MatchExpr( ArgT && arg, MatcherT const& matcher )
37 : ITransientExpression{ true, matcher.match( arg ) }, // not forwarding arg here on purpose
38 m_arg( CATCH_FORWARD(arg) ),
39 m_matcher( matcher )
40 {}

42 void streamReconstructedExpression( std::ostream& os ) const override {

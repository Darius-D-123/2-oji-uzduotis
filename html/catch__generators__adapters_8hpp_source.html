20 template <typename T>
21 class TakeGenerator final : public IGenerator<T> {
22 GeneratorWrapper<T> m_generator;
23 size_t m_returned = 0;
24 size_t m_target;
25 public:
26 TakeGenerator(size_t target, GeneratorWrapper<T>&& generator):
27 m_generator(CATCH_MOVE(generator)),
28 m_target(target)
29 {
30 assert(target != 0 && "Empty generators are not allowed");
31 }
32 T const& get() const override {
33 return m_generator.get();
34 }
35 bool next() override {
36 ++m_returned;
37 if (m_returned >= m_target) {
38 return false;
39 }
40
41 const auto success = m_generator.next();
42 // If the underlying generator does not contain enough values
43 // then we cut short as well
44 if (!success) {
45 m_returned = m_target;
46 }
47 return success;
48 }
49 };

57 template <typename T, typename Predicate>
58 class FilterGenerator final : public IGenerator<T> {
59 GeneratorWrapper<T> m_generator;
60 Predicate m_predicate;
61 static_assert(!std::is_reference<Predicate>::value, "This would most likely result in a dangling reference");
62 public:
63 template <typename P>
64 FilterGenerator(P&& pred, GeneratorWrapper<T>&& generator):
65 m_generator(CATCH_MOVE(generator)),
66 m_predicate(CATCH_FORWARD(pred))
67 {
68 if (!m_predicate(m_generator.get())) {
69 // It might happen that there are no values that pass the
70 // filter. In that case we throw an exception.
71 auto has_initial_value = next();
72 if (!has_initial_value) {
73 Detail::throw_generator_exception("No valid value found in filtered generator");
74 }
75 }
76 }
77
78 T const& get() const override {
79 return m_generator.get();
80 }
81
82 bool next() override {
83 bool success = m_generator.next();
84 if (!success) {
85 return false;
86 }
87 while (!m_predicate(m_generator.get()) && (success = m_generator.next()) == true);
88 return success;
89 }
90 };

98 template <typename T>
99 class RepeatGenerator final : public IGenerator<T> {
100 static_assert(!std::is_same<T, bool>::value,
101 "RepeatGenerator currently does not support bools"
102 "because of std::vector<bool> specialization");
103 GeneratorWrapper<T> m_generator;
104 mutable std::vector<T> m_returned;
105 size_t m_target_repeats;
106 size_t m_current_repeat = 0;
107 size_t m_repeat_index = 0;
108 public:
109 RepeatGenerator(size_t repeats, GeneratorWrapper<T>&& generator):
110 m_generator(CATCH_MOVE(generator)),
111 m_target_repeats(repeats)
112 {
113 assert(m_target_repeats > 0 && "Repeat generator must repeat at least once");
114 }
115
116 T const& get() const override {
117 if (m_current_repeat == 0) {
118 m_returned.push_back(m_generator.get());
119 return m_returned.back();
120 }
121 return m_returned[m_repeat_index];
122 }
123
124 bool next() override {
125 // There are 2 basic cases:
126 // 1) We are still reading the generator
127 // 2) We are reading our own cache
128
129 // In the first case, we need to poke the underlying generator.
130 // If it happily moves, we are left in that state, otherwise it is time to start reading from our cache
131 if (m_current_repeat == 0) {
132 const auto success = m_generator.next();
133 if (!success) {
134 ++m_current_repeat;
135 }
136 return m_current_repeat < m_target_repeats;
137 }
138
139 // In the second case, we need to move indices forward and check that we haven't run up against the end
140 ++m_repeat_index;
141 if (m_repeat_index == m_returned.size()) {
142 m_repeat_index = 0;
143 ++m_current_repeat;
144 }
145 return m_current_repeat < m_target_repeats;
146 }
147 };

154 template <typename T, typename U, typename Func>
155 class MapGenerator final : public IGenerator<T> {
156 // TBD: provide static assert for mapping function, for friendly error message
157 GeneratorWrapper<U> m_generator;
158 Func m_function;
159 // To avoid returning dangling reference, we have to save the values
160 T m_cache;
161 public:
162 template <typename F2 = Func>
163 MapGenerator(F2&& function, GeneratorWrapper<U>&& generator) :
164 m_generator(CATCH_MOVE(generator)),
165 m_function(CATCH_FORWARD(function)),
166 m_cache(m_function(m_generator.get()))
167 {}
168
169 T const& get() const override {
170 return m_cache;
171 }
172 bool next() override {
173 const auto success = m_generator.next();
174 if (success) {
175 m_cache = m_function(m_generator.get());
176 }
177 return success;
178 }
179 };

195 template <typename T>
196 class ChunkGenerator final : public IGenerator<std::vector<T>> {
197 std::vector<T> m_chunk;
198 size_t m_chunk_size;
199 GeneratorWrapper<T> m_generator;
200 bool m_used_up = false;
201 public:
202 ChunkGenerator(size_t size, GeneratorWrapper<T> generator) :
203 m_chunk_size(size), m_generator(CATCH_MOVE(generator))
204 {
205 m_chunk.reserve(m_chunk_size);
206 if (m_chunk_size != 0) {
207 m_chunk.push_back(m_generator.get());
208 for (size_t i = 1; i < m_chunk_size; ++i) {
209 if (!m_generator.next()) {
210 Detail::throw_generator_exception("Not enough values to initialize the first chunk");
211 }
212 m_chunk.push_back(m_generator.get());
213 }
214 }
215 }
216 std::vector<T> const& get() const override {
217 return m_chunk;
218 }
219 bool next() override {
220 m_chunk.clear();
221 for (size_t idx = 0; idx < m_chunk_size; ++idx) {
222 if (!m_generator.next()) {
223 return false;
224 }
225 m_chunk.push_back(m_generator.get());
226 }
227 return true;
228 }
229 };
